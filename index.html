تسجيل درجات الطلاب وسجلات التسليم في جدول student_results

غير مهيأ بعد (يعمل محلياً) 
create table if not exists public.student_results (
  id uuid primary key default gen_random_uuid(),
  student_name text not null,
  class_name text not null,
  exam_title text not null,
  score text not null,
  percentage integer not null default 0,
  grade text not null default '',
  elapsed_time text not null default '00:00',
  submitted_at text not null,
  created_at timestamptz not null default now()
);

alter table public.student_results enable row level security;

drop policy if exists "allow_insert_student_results" on public.student_results;
create policy "allow_insert_student_results"
on public.student_results
for insert
to anon, authenticated
with check (
  char_length(trim(student_name)) > 0
  and char_length(trim(class_name)) > 0
  and percentage between 0 and 100
);

drop policy if exists "allow_select_student_results" on public.student_results;
create policy "allow_select_student_results"
on public.student_results
for select
to anon, authenticated
using (true);
